---------------------------------------------------------------------
SECTION 4. UX RULES
---------------------------------------------------------------------
1. Landing page explains the project in under 10 seconds: what it is
   (hero + tagline), how it works (3 cards), proof (live demo), who
   made it (About). Two big buttons: ENTER THE LAB, PLAY TREASURE HUNT.
2. A persistent top nav (HOME / CIRCUIT LAB / TREASURE HUNT) with the
   active page highlighted.
3. Circuit Lab updates LIVE: any change to the circuit re-runs the
   simulation and refreshes diagram, histogram and the explainer.
   Errors show as a friendly red/green-neon message, never a traceback.
4. Every result has a short "WHAT JUST HAPPENED" line in plain English.
   Deeper math sits inside expanders titled "LEARN MORE".
5. Treasure Hunt: disabled boxes show an X after being opened; a
   persistent "ASK THE QUANTUM COMPUTER" button lets the player give
   up; after the round the reveal animation plays and Grover's result
   is compared with the player's guess count.
6. Keep the number of qubits at most 3 (fast, readable diagrams).
7. Empty states always say what to do next (e.g. "Pick a preset or
   click a gate to start").
8. Mobile: columns stack; buttons stay at least 44px tall; the hero
   text must not overflow at 390px width.
9. No external network calls except the Google Fonts stylesheet.
10. HARDWARE RUNS page: newest first, one card per saved image with
   backend, job ID and timestamp, and a caption saying honestly what
   the image shows. It only displays files saved earlier; it never
   triggers a run. The Circuit Lab offers a simulator-only "download
   run card" PNG.

# ---------- helpers ----------
def neon_hist(counts):
    items = sorted(counts.items())
    labels = [k for k, _ in items]
    total = sum(counts.values())
    vals = [v / total for _, v in items]
    fig, ax = plt.subplots(figsize=(6, 2.8))
    fig.patch.set_facecolor("#000000")
    ax.set_facecolor("#000000")
    bars = ax.bar(labels, vals, color=GREEN, edgecolor="#9dffb5", linewidth=1.2)
    for b, v in zip(bars, vals):
        ax.text(b.get_x() + b.get_width() / 2, v + 0.02, f"{v:.0%}",
                ha="center", va="bottom", color="#9dffb5", fontsize=8)
    ax.set_ylim(0, 1.15)
    ax.tick_params(colors=GREEN)
    for sp in ax.spines.values():
        sp.set_color(GREEN)
    ax.set_ylabel("probability", color=GREEN)
    fig.tight_layout()
    return fig


def draw_circuit(qc):
    try:
        fig = qc.draw("mpl", style="iqp-dark")
    except Exception:
        fig = qc.draw("mpl")
    return fig


def show_fig(fig):
    st.pyplot(fig)
    plt.close(fig)


def explain(qc, counts):
    total = sum(counts.values())
    ranked = sorted(counts.items(), key=lambda kv: -kv[1])
    top, top_n = ranked[0]
    ops = qc.count_ops()
    text = f"The most common result was {top} ({top_n / total:.0%} of 1,024 runs). "
    if len(counts) == 1:
        text += "Only one outcome appeared, so this circuit is deterministic."
    elif "h" in ops and "cx" in ops and len(counts) == 2:
        text += ("Two perfectly correlated outcomes is the signature of entanglement: "
                 "measuring one qubit tells you the other.")
    elif len(counts) == 2 ** qc.num_qubits:
        text += "All outcomes are possible: the qubits are in an even superposition."
    else:
        text += "Quantum interference made some outcomes likely and others rare."
    return text


def math_lines(qc):
    sv = Statevector(qc)
    n = qc.num_qubits
    lines = []
    for i, a in enumerate(sv.data):
        if abs(a) > 1e-9:
            lines.append(f"|{format(i, f'0{n}b')}>  amplitude {a.real:+.4f}{a.imag:+.4f}j   "
                         f"probability {abs(a) ** 2:.4f}")
    return "\n".join(lines)

def go(page):

def load_preset():
    n, text = PRESETS[st.session_state.preset]
    st.session_state.nq = n
    st.session_state.circuit_text = text


def add_line(line):
    cur = st.session_state.circuit_text.rstrip("\n")
    st.session_state.circuit_text = (cur + "\n" + line).strip()


def add_1q(g):
    add_line(f"{g} {st.session_state.t1}")


def add_2q(g):
    add_line(f"{g} {st.session_state.c2} {st.session_state.t2}")


def undo_line():
    lines = [l for l in st.session_state.circuit_text.splitlines() if l.strip()]
    st.session_state.circuit_text = "\n".join(lines[:-1])


def clear_text():
    st.session_state.circuit_text = ""


def new_round():
    s = st.session_state
    boxes = [format(i, f"0{s.g_n}b") for i in range(2 ** s.g_n)]
    s.g_secret = random.choice(boxes)
    s.g_opened = []
    s.g_done = False
    s.g_found = False
    s.g_guesses = 0


def change_mode():
    s = st.session_state
    s.g_n = 2 if s.g_mode.startswith("2") else 3
    s.g_rounds = 0
    s.g_total = 0
    new_round()


def guess(b):
    s = st.session_state
    s.g_opened.append(b)
    s.g_guesses += 1
    if b == s.g_secret:
        s.g_found = True
        s.g_done = True
        s.g_rounds += 1
        s.g_total += s.g_guesses


def give_up():
    st.session_state.g_done = True

def reveal_html(boxes, secret, opened):
    cells = []
    for i, b in enumerate(boxes):
        cls = "qbox treasure" if b == secret else "qbox"
        icon = "◆" if b == secret else ("✖" if b in opened else "·")
        cells.append(f'<div class="{cls}" style="animation-delay:{i * 0.12:.2f}s">'
                     f'<div class="qicon">{icon}</div>{b}</div>')
    return '<div class="box-grid">' + "".join(cells) + "</div>"


def game():
    s = st.session_state
    st.markdown("## > GROVER'S TREASURE HUNT")
    st.write("A treasure is hidden in one box. Click boxes to find it. Then Grover's quantum "
             "search algorithm finds it too, and we compare.")
    st.radio("Difficulty", ["2 qubits · 4 boxes", "3 qubits · 8 boxes"], key="g_mode",
             horizontal=True, on_change=change_mode)

    n = s.g_n
    boxes = [format(i, f"0{n}b") for i in range(2 ** n)]
    N = len(boxes)

    if not s.g_done:
        st.markdown("**Pick a box:**")
        for r in range(0, N, 4):
            for c, b in zip(st.columns(4), boxes[r:r + 4]):
                opened = b in s.g_opened
                c.button(f"✖ {b}" if opened else b, key=f"box_{b}", disabled=opened,
                         on_click=guess, args=(b,), use_container_width=True)
        st.button("ASK THE QUANTUM COMPUTER", key="giveup", on_click=give_up)
        st.caption(f"Guesses so far: {s.g_guesses}")
    else:
        st.markdown(reveal_html(boxes, s.g_secret, s.g_opened), unsafe_allow_html=True)
        counts = run_sim(grover_circuit(s.g_secret))
        top = max(counts, key=counts.get)
        p = counts[top] / sum(counts.values())
        k = grover_iterations(n)
        if s.g_found:
            you = f"You found it in {s.g_guesses} guess(es)."
        else:
            you = "You asked the quantum computer."
        st.markdown(
            f'<div class="card"><h4>RESULT</h4><p>{you} Grover\'s algorithm answered '
            f'<b>{top}</b> using {k} oracle call(s), with {p:.0%} confidence. '
            f'A classical search needs about {(N + 1) / 2:.1f} guesses on average.</p></div>',
            unsafe_allow_html=True)
        show_fig(neon_hist(counts))
        st.button("▶ NEW ROUND", key="newround", on_click=new_round)
        with st.expander("LEARN MORE: why Grover is faster"):
            st.write(f"Searching N unsorted items takes about N/2 guesses classically but only "
                     f"about (pi/4)·sqrt(N) oracle calls with Grover. Here N = {N}, so {k} call(s). "
                     "The oracle flips the sign (phase) of the target; the diffusion step reflects "
                     "every amplitude about the average, boosting the target and shrinking the rest.")
            show_fig(draw_circuit(grover_circuit(s.g_secret)))

    avg = (s.g_total / s.g_rounds) if s.g_rounds else 0
    st.caption(f"Rounds found: {s.g_rounds} | Your average guesses: {avg:.2f} | "
               f"Classical average: {(N + 1) / 2:.1f}")


def runs_gallery():
    st.markdown("## > HARDWARE RUNS")

            with open(jf, encoding="utf-8") as f:
                m = json.load(f)
            m["_path"] = os.path.join(RUNS_DIR, m.get("image", ""))

                "then commit docs/hardware_runs/.")

        st.markdown(

            f'<p class="small">backend {html.escape(str(m.get("backend", "?")))} | '
            f'job {html.escape(str(m.get("job_id", "?")))} | '
            f'{html.escape(str(m.get("timestamp", "")))}</p>'
            f'<p>{html.escape(str(m.get("caption", "")))}</p></div>',
            unsafe_allow_html=True)
        st.image(m["_path"], use_container_width=True)

SECTION 8. KNOWN PITFALLS (check these when something looks wrong)
---------------------------------------------------------------------
1. ICON FONTS BREAK: do not apply font-family to "*" or to every
   span. Streamlit's expander arrows use an icon font; forcing
   JetBrains Mono on them shows raw words like "keyboard_arrow_down".
   The CSS above targets only text elements for that reason.
2. MATRIX RAIN NOT SHOWING: the rain is injected into the parent page
   from a zero-height component. If the browser blocks it, or a
   Streamlit update changes the iframe sandbox, the app must still
   work. Fallback: delete the components.html(RAIN_HTML ...) line and
   add a CSS-only background such as a subtle repeating-linear-gradient
   of green scanlines. Never let the rain block clicks (it uses
   pointer-events:none).
3. DEPRECATION WARNINGS: newer Streamlit versions may deprecate
   use_container_width (replacement: width="stretch") or
   components.html. Run "pip show streamlit", read the warning text,
   and apply the replacement from the official docs
   (docs.streamlit.io). Keep behavior identical.
4. BUTTON STYLE SELECTORS: Streamlit changes internal test ids and
   attributes between versions. If primary (active nav) buttons are
   not green, inspect the element in the browser dev tools and adjust
   the .stButton > button[...] selectors.
5. SESSION STATE ERRORS: never assign to a widget's key AFTER the
   widget is drawn in the same run. This file changes widget state
   only inside on_click / on_change callbacks, which is the safe way.
   Keep it that way when you add features.
6. DIAGRAM STYLE: qc.draw("mpl", style="iqp-dark") needs matplotlib
   and pylatexenc. If the style name is rejected in your Qiskit
   version, the code falls back to the default style.
7. QUBIT ORDER CONFUSION: the Lab explains that qubit 0 is the
   rightmost printed bit. Keep that sentence visible.
8. PERFORMANCE: 3 qubits x 4096 shots is instant on the simulator;
   do not raise the qubit limit above 3 without testing.
9. GALLERY EMPTY AFTER DEPLOY: the gallery reads docs/hardware_runs/
   relative to the repo root. The PNG and JSON files must be committed
   and pushed, and not matched by .gitignore.
10. IMAGE PARAMETER DEPRECATION: st.image(use_container_width=True)
   may warn in newer Streamlit; switch to the documented replacement.
11. The gallery uses html.escape on captions. Keep it that way.
